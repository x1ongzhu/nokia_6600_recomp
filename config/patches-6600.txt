# Bring-up text patches for the flat 6600 runtime image.
#
# Format: <runtime VA> <hex bytes>  # explanation
# Applied by: recompiler/romimage.py build --patch config/patches-6600.txt
#
# These are NOT firmware modifications.  They restore bytes that the packed
# flash image does not contain (or that are overwritten by the next packed
# section) so the boot harness can keep executing.  Every patch is printed by
# the build and recorded in the layout JSON.

# euser.dll Thumb helper at 0x5039ad9e: the visible function ends with
# `movs r4, r2` at 0x5039adca, but the following bytes (0x5039adcc) are the
# start of a packed data/jump table, so the `pop {r4, r7, pc}` tail is absent.
# Restore the natural epilogue.  Without this the caller runs into the data and
# the whole DLL2 call is skipped by NK_RECOVER.
0x5039adcc 90bd
