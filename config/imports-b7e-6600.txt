# imports-b7e-6600.txt - bring-up rebinds for the kernel's EUser imports
#
# Syntax: <slot-va> <target-va|zero>
#
# A flat runtime image already binds every real export (the slot *is* the
# exporting image's export-table entry).  The kernel, however, was linked
# against an EUser .def revision whose ordinal order differs from the EUser
# DLL in this ROM.  The four slots below resolve to a function with the wrong
# ABI and panic during kernel init:
#
#   0x503b391c  EUser ord  98  -> 0x5038e594  (array accessor)
#               caller: kernel panic printer, passes a TPtrC16 ("USER")
#               want:   TDesC16::AtC          -> 0x50388474
#
#   0x503b4d1c  EUser ord 1378  -> 0x50388104  (3-arg TPtr16 ctor)
#               caller: kernel panic init, passes (buf=0, len=0, max=-1)
#               want:   2-arg TPtr16 ctor     -> 0x503880ec
#
#   0x503b4b9c  EUser ord 1282  -> 0x50388a2c  (TDes8::Copy(const TUint8*))
#               caller: object-container init, passes (this, 8)
#               want:   TBuf16(int)           -> 0x50388254
#
#   0x503b3b70  EUser ord  247  -> 0x50388408  (TDes16::Copy(const TDesC16&))
#               caller: locale init, passes (this, pointer table, 0)
#               want:   TDes16::Copy(const TUint16*, TInt) -> 0x503883c0
#               (3-arg overload: r2=0 copies zero characters; the 2-arg
#                string overload scans the pointer table and panics USER 12)
#
# These are explicit bring-up rebinds, not recovered ROM state.  The test
# tools/test_b7e.sh pins them; the confirmation-only import map stays in
# config/imports-6600.txt.
0x5037d0e8 zero

0x503b4d1c 0x503880ec
0x503b4b9c 0x50388254
